"use client";

import { useState, useEffect } from "react";

// Add this to your existing states:
const [remainingSearches, setRemainingSearches] = useState<number>(100);

// Update your searchWithBrowserless function:
const searchWithBrowserless = async (pageNumber: number = 1, maxRetries = 3) => {
  if (!keyword || !pickedLocation) {
    setStatus("⚠️ Please enter keyword and select location");
    return;
  }

  setLoading(true);
  setStatus("🔄 Starting search...");

  // Add delay
  const delay = 5000 + Math.random() * 5000;  // 5-10 seconds
  setStatus(`⏳ Waiting ${Math.round(delay/1000)}s...`);
  await sleep(delay);

  // Retry logic
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      setStatus(`🔄 Attempt ${attempt}/${maxRetries}...`);
      
      const params = new URLSearchParams({
        q: keyword,
        city: pickedLocation.city,
        state: pickedLocation.state || "",
        country: pickedLocation.countryCode,
        page: pageNumber.toString(),
        hl: language,
        domain,
        adtest: "on",
      });

      const response = await fetch(`/api/search-browserless?${params.toString()}`);
      const data = await response.json();

      // ✅ UPDATE REMAINING SEARCHES
      if (data.remaining !== undefined) {
        setRemainingSearches(data.remaining);
      }

      if (data.success) {
        searchCount++;
        setStatus(
          `✅ Success via ${data.proxyUsed}! ` +
          `(${data.remaining} searches left this hour)`
        );
        
        window.open(data.googleUrl, "_blank", "noopener,noreferrer");
        
        setTimeout(() => setStatus(""), 5000);
        setLoading(false);
        return;
      }

      if (data.blocked || data.rateLimited) {
        const waitSec = data.retryAfterSec || 30;
        
        if (data.remaining === 0) {
          // Hourly limit reached
          setStatus(
            `⚠️ Hourly limit reached (100 searches). ` +
            `Resets in ${Math.ceil(waitSec/60)} minutes. ` +
            `Opening in browser...`
          );
          window.open(data.googleUrl, "_blank", "noopener,noreferrer");
          setLoading(false);
          return;
        }
        
        if (attempt < maxRetries) {
          setStatus(`⚠️ ${data.error} - Retrying in ${waitSec}s...`);
          await sleep(waitSec * 1000);
          continue;
        } else {
          setStatus(`⚠️ ${data.error} - Opening in browser`);
          window.open(data.googleUrl, "_blank", "noopener,noreferrer");
        }
      }

    } catch (error: any) {
      console.error(`❌ Attempt ${attempt} failed:`, error);
      
      if (attempt === maxRetries) {
        setStatus("❌ API unavailable - Opening in browser");
        openGoogle(pageNumber);
      } else {
        setStatus(`⚠️ Error - Retrying...`);
        await sleep(5000);
      }
    }
  }

  setLoading(false);
  setTimeout(() => setStatus(""), 5000);
};
